# 5. ERROR RESPONSE EXAMPLE - Invalid Token
# ============================================================================

# Request with bad token:
curl -X POST http://localhost:8000/api/v1/integration/workout/generate \
  -H "Authorization: Bearer invalid-token" \
  -H "Content-Type: application/json" \
  -d '{
    "goal": "muscle_gain",
    "duration_weeks": 12,
    "frequency": 4,
    "equipment": ["dumbbell"],
    "intensity": "high"
  }'

# Response (HTTP 401):
{
  "detail": "Invalid or expired token"
}

# Frontend handling:
// Clear localStorage and redirect
localStorage.removeItem('access_token');
localStorage.removeItem('refresh_token');
window.location.href = '/login';

# 9. ASYNC PROCESSING EXAMPLE (For long-running requests)
# ============================================================================

# Start async task:
curl -X POST http://localhost:8000/api/v1/integration/workout/generate/async \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "goal": "muscle_gain",
    "duration_weeks": 12,
    "frequency": 4,
    "equipment": ["dumbbell", "barbell"],
    "intensity": "high"
  }'

# Response (HTTP 202 Accepted):
{
  "status": "queued",
  "task_id": "task-id-uuid",
  "status_url": "/api/v1/integration/status/task-id-uuid"
}

# Poll for status:
curl -X GET http://localhost:8000/api/v1/integration/status/task-id-uuid \
  -H "Authorization: Bearer $TOKEN"

# Response (HTTP 200) - Still processing:
{
  "task_id": "task-id-uuid",
  "status": "PENDING",
  "result": null,
  "progress": {
    "current": 3,
    "total": 6,
    "current_step": "calling_ai"
  }
}

# Poll again after 5 seconds:
